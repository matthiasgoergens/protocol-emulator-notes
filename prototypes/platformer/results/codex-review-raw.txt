(* PE-X: the generic 16-bit PE of ../pe-synth (pe16) with four generic extensions, as an
   executable specification. pex.ml is the RTL and must match this cycle for cycle.

   pe16, unchanged: a 16-bit state s, updated when a step arrives by
     s <- sat (op x y), op in {add, sub, max, min} (signed, saturating),
     x = s or the neighbour's data, y = the constant k or s,
   a byte-wide configuration chain, and a pipeline register passing the neighbour's word on.
   pe16's "en" input becomes a tag on the word itself (a step is a word tagged STEP).

   The extensions:
   1. A tagged link. Each word carries a 3-bit tag: bits 1:0 = 0 none, 1 step, 2 record word,
      3 last word of a record (like AXI-stream's TLAST); bit 2 = a record class.
   2. Record loading ("the first empty cell takes it"). A PE with load_en set and not loaded
      consumes record words of its class as they pass (they leave as tag none) and writes them to
      R0 attr, R1 a[31:16], R2 a[15:0], R3 s, R4 attr2 (9 bits) in that order; the last word sets
      loaded. A PE releases (loaded <- 0) on a step whose data has bit 15 set (end of run) if
      rel_eol, or on the first step outside its window after one inside it if rel_win.
   3. A window predicate on s: p = ((s >> w) & (2^n - 1)) = 0, so n = 0 means always.
   4. A 32-bit shift register a, shifting (or rotating) by b = 1, 2, 4 or 8 bits on every step
      inside the window while loaded, MSB first, or LSB first when attr2 bit 8 (dir) is set. The
      field shifted out is the top (or bottom) b bits.
   5. A conditional write into the passing word, split into an 8-bit value lane (bits 7:0) and an
      8-bit flag lane (bits 15:8): on a step with write_en, loaded, p, field <> 0 and
      (flags & attr2[7:0]) = 0, value <- attr[7:0] | field and flags <- flags | attr[15:8].

   Configuration bytes (shifted in first to last):
     c0: [1:0] alu op, [2] y = k (else s), [3] x = s (else neighbour data), [4] step s,
         [5] load_en, [6] class, [7] rotate a (else shift in zeros)
     c1, c2: k, high byte first

type tile = { pix : int array; (* 16 pixels, 0-3, 0 transparent *) pal : int; front : bool }

(* the sprites the chip is given: on-screen ones, at most nspr, the highest-priority ones kept *)
let visible_sprites l =
  let v = List.filter (fun s -> s.x > -16 && s.x < npix && Array.exists (( <> ) 0) s.spix) l.sprites in
  let n = List.length v in
  if n > nspr then List.filteri (fun i _ -> i >= n - nspr) v else v

(* Reference: what each pixel must show, straight from the description. *)
let reference l =
  Array.init npix (fun px ->
    let xs = px + l.fine in
    let t = l.tiles.(xs / 16) in
    let v = t.pix.(xs mod 16) in
    let c = ref l.backdrop and opaque = ref false and front = ref false in
    if v <> 0 then (c := t.pal * 4 + v; opaque := true; front := t.front);
    List.iter (fun s ->
      let d = px - s.x in
      if d >= 0 && d < 16 then begin
        let v = s.spix.(if s.flip then 15 - d else d) in
        if v <> 0 && not !front && not (s.behind && !opaque) then c := 16 + s.spal * 4 + v
      end) (visible_sprites l);
    !c)

let t_word1 = 6 and t_last1 = 7

let record ~cls words =
  let n = List.length words in
  List.mapi (fun i w ->
    let last = i = n - 1 in
    ((if cls = 1 then (if last then t_last1 else t_word1) else if last then t_last else t_word), w land 0xFFFF))

(* a lookup-table update: start address, then the entries *)

let encode ?(lut = []) l =

  let first_tiles = List.concat (List.init kcells (fun k ->
    let t = l.tiles.(k) in
    let a = pattern t.pix in
    let a = if k = 0 then (a lsl (2 * l.fine)) land m32 else a in
    record ~cls:0 [ tile_attr t; a lsr 16; a; l.fine - 16 * k; 0 ])) in
  let sprites = List.concat_map (fun s ->
    let a = pattern s.spix in

    let attr2 = (if s.behind then 0x03 else 0x02) lor (if s.flip then 0x100 else 0) in
    record ~cls:0 [ 16 + s.spal * 4; a lsr 16; a; -s.x; attr2 ]) (visible_sprites l) in
  let later = List.concat (List.init (ntiles - kcells) (fun i ->
    let j = i + kcells in
    let t = l.tiles.(j) in
    let a = pattern t.pix in
    (t_wait, 16 * (j - kcells + 1)) :: record ~cls:0 [ tile_attr t; a lsr 16; a ])) in
  [ (t_pixels, l.backdrop) ] @ first_tiles @ sprites @ lut @ later @ [ (t_end, 0) ]

  let shifted_right = mux lb (List.map (fun b -> mux2 rot (rotr av b) (srl av b)) bits_list) in

  let write = is_step &: write_en &: loaded.value &: p &: (field <>:. 0) &: ~:blocked in
  let written = concat_msb [ flags |: select attr.value 15 8; select attr.value 7 0 |: field ] in
  let eol = bit data 15 in
  let release =
    if fault = 3 then loaded.value &: ((rel_eol &: eol) |: (rel_win &: ~:(pprev.value) &: p))
    else loaded.value &: ((rel_eol &: eol) |: (rel_win &: pprev.value &: ~:p)) in

    ; otag <-- mux2 consume (zero 3) tag
    ; when_ is_step

        ; when_ (loaded.value &: p) [ a <-- mux2 dir shifted_right shifted_left ]
        ; when_ release [ loaded <-- gnd ]
        ; pprev <-- mux2 eol gnd p ]
    ; when_ consume
        [ switch widx.value
            [ of_int ~width:3 0, [ attr <-- data ]
            ; of_int ~width:3 1, [ a <-- concat_msb [ data; select av 15 0 ] ]
            ; of_int ~width:3 2, [ a <-- concat_msb [ select av 31 16; data ] ]
            ; of_int ~width:3 3, [ s <-- data ]

        ; if_ (kind ==:. 3) [ loaded <-- vdd; widx <--. 0 ]
            [ when_ (widx.value <>:. 7) [ widx <-- widx.value +:. 1 ] ] ] ];
  { tag = otag.value; data = odata.value; s = s.value; cfg_out }

(* a row of n PE-Xs, the link and the configuration chain running through them *)
let row ?(fault = 0) ?(lean = false) n () =
  let clock = input "clock" 1 and clear = input "clear" 1 in
  let cfg_in = input "cfg_in" 8 and cfg_strobe = input "cfg_strobe" 1 in
  let tag = ref (input "tag_in" 3) and data = ref (input "data_in" 16) and cfg = ref cfg_in in
  let outs = List.concat (List.init n (fun i ->
    let o = create ~fault ~lean ~clock ~clear ~cfg_in:!cfg ~cfg_strobe ~tag:!tag ~data:!data () in
    tag := o.tag; data := o.data; cfg := o.cfg_out;
    [ output (Printf.sprintf "s%d" i) o.s ])) in
  Circuit.create_exn ~name:(Printf.sprintf "pex_row%d" n)
    (outs @ [ output "tag_out" !tag; output "data_out" !data; output "cfg_out" !cfg ])

## Confirmed drawing mismatch

**Left-clipped flipped sprites are encoded incorrectly.** In [scene.ml](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/scene.ml:56), sprites with `x < 0` are shifted right to clip them. The PE then reads the shifted pattern LSB first for a flipped sprite. But the reference renderer selects pixel `15 - d` at screen position `d = px - x` ([scene.ml](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/scene.ml:33)).

For example, make a sprite at `x = -1`, set `flip = true`, and set only `spix.(15) = 1`. The reference shows that pixel at screen `x = 0`. The encoder shifts the pattern right by two bits, moving pixel 15 out of the LSB-first sequence; the chip shows transparency there. This is reachable through `Scene.encode`. A focused check with an asymmetric sprite at negative x would expose it; random testing may hit such cases, but its finite run and broad random coverage don’t establish this case independently.

The other listed edge cases look substantially better covered by inspection and by the random checker:

- `fine` is sampled across 0–15; the reference indexes the 17-tile array through `px + fine`, and the encoder emits the additional slice needed at the right edge.
- Sprite x is sampled from −20 to 269, including both clipping edges; lists can contain up to 21 sprites, and `visible_sprites` retains the highest-priority 16.
- Blank lines are included in the simulated field, but the chip comparison records only visible lines. That leaves room to test loaded-state transitions directly across visible/blank sequences; the encoder’s EOL step should release loaded cells at the end of each drawn line.
- The first simulated field starts from reset and includes blank lines before the first visible line. That exercises a reset-to-visible transition, though it does not prove every unusual host packet sequence safe.
- LUT class-1 records are inserted ahead of the line packet and are intended to pass through class-0 cells to the output port. The checker’s 0-mismatch result supports that path for its generated packets.

The checker reports 240 random lines, zero mismatches, and three planted faults caught ([results/check.txt](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/results/check.txt)). That is useful evidence for the exercised cases; it does not cover every possible accepted `Scene.encode` input or arbitrary packet ordering.

## Generic PE claim

The claim is **substantially honest at the cell level**: tile and sprite behaviour comes from the same PE-X configuration, and the lockstep PE model test reports zero mismatching cycles over 300 random runs ([results/pecheck.txt](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/results/pecheck.txt)). The surrounding chip is still video-specific: the feeder interprets `WAIT`, `PIXELS`, and `END`, and the output port implements the colour table, pixel hold, burst selection, and NCO ([video.ml](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/video.ml:19), [video.ml](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/video.ml:43)).

The least convincing “generic” extension is the **conditional write into the passing word**. Its field extraction followed by OR-ing into an 8-bit value lane, with a separate 8-bit flag lane and mask, is a pixel-compositing rule in disguise ([pex_model.ml](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/pex_model.ml:47)). A smaller reusable primitive would be a generic masked merge: select a configurable field, then replace bits selected by a mask. That would avoid baking pixel opacity and palette composition into the PE, while retaining sprite priority and tile-front behaviour.

## Streaming versus a tile cache

The line-buffer arithmetic is internally consistent: two banks of 256 × 19 bits are 9,728 bits, or about 40,110 µm² at the stated 4.12 µm²/bit. The packet peak is 157 entries per line at 10 clocks each, 1,570 of 3,404 line clocks, so the stated 46% host-link occupancy has room to spare ([budget.py](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/budget.py:20), [results/budget.txt](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/results/budget.txt)).

The comparison is **not enough to establish that streaming is better than an on-chip cache**. Under the same area estimate, 64 thick-gain-cell tiles cost about 135,108 µm² and 128 cost about 270,216 µm², versus about 40,110 µm² for the line buffer. The line buffer is smaller, and its assumed 3.1 ms retention comfortably exceeds its 128 µs maximum data lifetime. But the cache’s stated refresh load—0.78% and 1.55% of a 32-bit port—is also modest. Whether that cache works depends on the memory’s refresh/read timing and peripheral design; the budget compares area and an estimated port rate, not a complete cache implementation against the streamer. It also counts “tile patterns” for the cache but full tagged line entries for the buffer, and omits the tile map or indices from the cache total.

## Area comparison

The reported 100,432 µm² for the retro console is **not an apples-to-apples total against the proposed video chip**. Summing the supplied platformer blocks gives approximately:

- 18 PE-X cells: 18 × 18,110 = **325,980 µm²**
- feeder: **5,994 µm²**
- output port: **5,305 µm²**
- sequencer: **18,995 µm²**

That is about **356,274 µm²**, before accounting for the line buffer’s memory implementation. Even the lean PE-X figure gives a PE-row area of about 258,534 µm², and roughly 288,828 µm² for those same blocks. These are block totals from separate reports, so treat them as an approximate comparison; still, neither supports a claim that the full programmable chip is smaller than the 100,432 µm² console.

The full and lean PE-X figures also describe different designs: the lean variant fixes the window and shift settings used by this video configuration and removes rotation support ([pex.ml](/home/matthias/prog/janestreet/emulator-wt-platformer/prototypes/platformer/pex.ml:32)). It is fair to report both as a general PE-X and a video-specialised variant, but not to present the lean number as the area of the fully generic PE-X.

**Where the design holds up:** the configuration-based tile/sprite datapath is real, the packet budget has clear headroom, and the random lockstep checker plus planted faults are meaningful evidence. The negative-x flipped-sprite case is a concrete counterexample to renderer equivalence, and the area comparison needs to distinguish the PE-X cell, the whole video chip, and the specialised console.
