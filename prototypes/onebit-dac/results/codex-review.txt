    78	        "snr_idle_a": 10 * np.log10(fs_pow / nza),

   146	    shared = dc.Impair(sj=30e-12, delta=100e-12, sv_inband=3e-6, kpads=8, bmax=0.2, kd=1e-9)

   148	             ("o3", "B", [("ideal", dc.Impair()), ("quietrail", quiet), ("sharedrail", shared)]),

   152	        wr, _ = host(mode, R, order)

   155	            _, yl, fsd = dc.analogue(bl, imp)
   156	            _, yr, _ = dc.analogue(br, dc.Impair(**{**imp.__dict__, "seed": imp.seed + 100}))
   157	            # to 48 kHz for listening: 93,750 -> 48,000 is 128/250
   158	            l48 = resample_poly(yl, 128, 250)
   159	            r48 = resample_poly(yr, 128, 250)
   160	            name = f"music_{order}_mode{mode}_{short}.wav"
   161	            dc.write_wav(os.path.join(WAV, name), l48, r48, 48000)
   162	            lines.append(f"{order} mode {mode} {imp.label():<60} -> wav/{name}")
   163	            print(lines[-1], flush=True)
   164	    # underruns: 8 gaps of 20 ms at random places. The chip holds the last frame (what the pump
   165	    # does); the alternative substitutes zeros. Click measure, per gap edge: the peak of the
   166	    # output high-passed above 4 kHz within +-1 ms of the edge, in dB re the same measure on the
   167	    # gapless output at the same place in the music (0 dB: the edge adds nothing audible above
   168	    # the music's own transients there).
   169	    from scipy import signal as ss
   170	    rng = np.random.default_rng(21)
   171	    n_gap = int(0.02 * dc.FS_A)
   172	    starts = np.sort(rng.choice(np.arange(int(0.5 * dc.FS_A), int(5.0 * dc.FS_A)), 8, replace=False))
   173	    hpf = ss.butter(4, 4000, "high", fs=dc.FM / 64, output="sos")
   174	    w0, steps0 = host("A", L, "o3")
   175	    b0, _ = run("o3", w0, w0, steps0, "underrun_none")
   176	    _, y0, fsd0 = dc.analogue(b0, dc.Impair())
   177	    hp0 = ss.sosfilt(hpf, y0)
   178	    ms = int(0.001 * fsd0)
   179	    for policy in ["hold", "zero"]:
   180	        Lu, Ru, gaps, prev = [], [], [], 0
   181	        shift = 0
   182	        for st in starts:
   183	            Lu.append(L[prev:st]); Ru.append(R[prev:st])
   184	            Lu.append(np.full(n_gap, L[st - 1] if policy == "hold" else 0.0))
   185	            Ru.append(np.full(n_gap, R[st - 1] if policy == "hold" else 0.0))
   186	            gaps.append(st + shift)
   187	            shift += n_gap
   188	            prev = st
   189	        Lu.append(L[prev:]); Ru.append(R[prev:])
   190	        Lu, Ru = np.concatenate(Lu), np.concatenate(Ru)
   191	        wl, steps = host("A", Lu, "o3")
   192	        wr, _ = host("A", Ru, "o3")
   193	        bl, br = run("o3", wl, wr, steps, f"underrun_{policy}")
   194	        _, yl, fsd = dc.analogue(bl, dc.Impair())
   195	        _, yr, _ = dc.analogue(br, dc.Impair())
   196	        dc.write_wav(os.path.join(WAV, f"underrun_{policy}_o3_modeA.wav"), resample_poly(yl, 128, 250), resample_poly(yr, 128, 250), 48000)
   197	        hp = ss.sosfilt(hpf, yl)
   198	        clicks = []
   199	        for st, g in zip(starts, gaps):
   200	            ref = np.max(np.abs(hp0[int(st / dc.FS_A * fsd) - ms : int(st / dc.FS_A * fsd) + ms]))
   201	            for e in [g, g + n_gap]:
   202	                c = int(e / dc.FS_A * fsd)
   203	                clicks.append(20 * np.log10(np.max(np.abs(hp[c - ms : c + ms])) / ref))
   204	        lines.append(f"underrun policy {policy}: 8 gaps of 20 ms, 16 edges; high-passed (>4 kHz) peak within 1 ms of an edge, "
   205	                     f"dB re the gapless output there: median {np.median(clicks):+.1f}, worst {np.max(clicks):+.1f} -> wav/underrun_{policy}_o3_modeA.wav")
   206	        print(lines[-1], flush=True)
   207	    with open(os.path.join(RES, "music.txt"), "w") as f:
   208	        f.write("\n".join(lines) + "\n")

    48	# ---------------- analogue model ----------------
    49	# The pin output per modulator step is represented by its area (volt-seconds) over the step;
    50	# for audio-band results only the area matters, and every impairment below is an area error.
    51	#  - level: the pin is high (V_hi) or low (0). The stored bit F is 1 for y = -1, so the pin
    52	#    drives NOT F (the pin stage's invert, or the host negates the words).
    53	#  - supply: V_hi = V0 + n_s, n_s low-pass noise (one pole at 100 kHz) given by its rms in
    54	#    20 Hz-20 kHz (the IO rail of the audio pins).
    55	#  - rise/fall asymmetry: a rising edge loses V*tr_eff of area, a falling edge gains V*tf_eff;
    56	#    with tr_eff = t0 - delta/2 and tf_eff = t0 + delta/2 the t0 parts cancel per pulse and
    57	#    every edge adds V*delta/2: an error proportional to the number of transitions.
    58	#  - jitter: an edge displaced by dt changes the area by -/+ V*dt (rising/falling), dt ~ N(0, sj).
    59	#  - aggressor bounce: k_pads pads on the same rail switching with random data at the audio
    60	#    pin's edges move each edge by kd * (bounce voltage), the bounce being bmax * (number
    61	#    switching - k/2)/(k/2) (zero mean); kd in s per volt.
    62	
    63	
    64	@njit(cache=True)
    65	def _edge_dt(sj, kpads, bmax, kd):
    66	    dt = 0.0
    67	    if sj > 0:
    68	        dt += sj * np.random.standard_normal()
    69	    if kpads > 0:
    70	        ns = 0
    71	        for _ in range(kpads):
    72	            if np.random.random() < 0.5:
    73	                ns += 1
    74	        dt += kd * bmax * (ns - kpads / 2) / (kpads / 2)
    75	    return dt
    76

    78	@njit(cache=True)
    79	def _areas(bits, T, V0, delta, sj, sv_step, kpads, bmax, kd, seed, mode):
    80	    """mode 0: NRZ on one pin; 1: NRZ differential (pin and its complement, same clock edge,
    81	    output = difference / 2); 2: RZ on one pin (a 1 is high for the first half step)."""
    82	    np.random.seed(seed)
    83	    n = bits.shape[0]
    84	    a = np.empty(n, dtype=np.float64)
    85	    prev = 1 - bits[0]
    86	    # supply noise: white noise through one pole at 100 kHz (regulator noise and hum are
    87	    # low-frequency; sv_step is the rms of this process)
    88	    al = np.exp(-2 * np.pi * 100e3 * T)
    89	    ns = 0.0
    90	    for i in range(n):
    91	        b = 1 - bits[i]
    92	        if sv_step > 0:
    93	            ns = al * ns + np.sqrt(1 - al * al) * sv_step * np.random.standard_normal()
    94	        vhi = V0 + ns
    95	        if mode == 2:
    96	            area = 0.0
    97	            if b == 1:
    98	                dt1 = _edge_dt(sj, kpads, bmax, kd)
    99	                dt2 = _edge_dt(sj, kpads, bmax, kd)
   100	                # every edge adds V*delta/2 (rise/fall asymmetry); a late rise loses, a late fall gains
   101	                area = vhi * T / 2 + V0 * (delta + dt2 - dt1)
   102	            a[i] = area
   103	            prev = b
   104	            continue
   105	        d = b - prev
   106	        dt = _edge_dt(sj, kpads, bmax, kd) if d != 0 else 0.0
   107	        e = 0.0
   108	        if d != 0:
   109	            e = V0 * (0.5 * delta - d * dt)
   110	        if mode == 0:
   111	            a[i] = T * vhi * b + e
   112	        else:
   113	            # complement pin: same supply sample, opposite edge, same asymmetry contribution
   114	            ec = 0.0
   115	            if d != 0:
   116	                ec = V0 * (0.5 * delta + d * dt)
   117	            a[i] = 0.5 * ((T * vhi * b + e) - (T * vhi * (1 - b) + ec)) + 0.5 * T * V0
   118	        prev = b
   119	    return a

   144	# unity-gain Sallen-Key Butterworth at 40 kHz (R 10 kohm, 10 kohm; C 560 pF feedback,

   150	def filter_sos():
   151	    w1 = 1 / (R1 * C1)
   152	    w0 = 1 / (RS * np.sqrt(CA * CB))
   153	    q = np.sqrt(CA * CB) / (2 * CB)
   154	    num = [w1 * w0 ** 2]
   155	    den = np.polymul([1, w1], [1, w0 / q, w0 ** 2])
   156	    b, a = signal.bilinear(num, den, fs=FM)
   157	    return signal.tf2sos(b, a)

   160	def analogue(bits, imp, dec=64):
   161	    """pin areas -> filter (output in volts, as the pin's average) -> decimated by dec"""
   162	    sos = filter_sos()
   163	    x = imp.areas(bits) * FM          # volts, step average
   164	    y = signal.sosfilt(sos, x)
   165	    # decimate to FM/dec with an FIR well below the new Nyquist
   166	    yd = signal.resample_poly(y, 1, dec, window=("kaiser", 14.0))
   167	    return y, yd, FM / dec

   177	def spectrum(x, fs):
   178	    x = x - np.mean(x)
   179	    w = signal.windows.kaiser(len(x), 38)
   180	    X = np.fft.rfft(x * w)
   181	    # Parseval: summed over the bins of a component, p gives its mean square (A^2/2 for a sine)
   182	    p = 2 * np.abs(X) ** 2 / (len(x) * np.sum(w ** 2))
   183	    f = np.fft.rfftfreq(len(x), 1 / fs)
   184	    return f, p, None

   187	def tone_metrics(x, fs, f0, nb=40, weight=False):
   188	    """returns (signal rms^2, THD+N rms^2 in 20 Hz-20 kHz, list of harmonic levels in dB re signal)"""
   189	    f, p, _ = spectrum(x, fs)
   190	    df = f[1] - f[0]
   191	    k0 = int(round(f0 / df))
   192	    sig = p[k0 - nb : k0 + nb + 1].sum()
   193	    band = (f >= 20) & (f <= 20000)
   194	    mask = band.copy()
   195	    mask[k0 - nb : k0 + nb + 1] = False
   196	    pw = p * (a_weight(np.maximum(f, 1)) ** 2 if weight else 1.0)
   197	    noise = pw[mask].sum()
   198	    harms = []
   199	    for h in range(2, 6):
   200	        kh = int(round(h * f0 / df))
   201	        if h * f0 <= 20000:
   202	            harms.append(10 * np.log10(p[kh - nb : kh + nb + 1].sum() / sig))
   203	    return sig, noise, harms

    48	# ---------------- analogue model ----------------
    49	# The pin output per modulator step is represented by its area (volt-seconds) over the step;
    50	# for audio-band results only the area matters, and every impairment below is an area error.
    51	#  - level: the pin is high (V_hi) or low (0). The stored bit F is 1 for y = -1, so the pin
    52	#    drives NOT F (the pin stage's invert, or the host negates the words).
    53	#  - supply: V_hi = V0 + n_s, n_s low-pass noise (one pole at 100 kHz) given by its rms in
    54	#    20 Hz-20 kHz (the IO rail of the audio pins).
    55	#  - rise/fall asymmetry: a rising edge loses V*tr_eff of area, a falling edge gains V*tf_eff;
    56	#    with tr_eff = t0 - delta/2 and tf_eff = t0 + delta/2 the t0 parts cancel per pulse and
    57	#    every edge adds V*delta/2: an error proportional to the number of transitions.
    58	#  - jitter: an edge displaced by dt changes the area by -/+ V*dt (rising/falling), dt ~ N(0, sj).
    59	#  - aggressor bounce: k_pads pads on the same rail switching with random data at the audio
    60	#    pin's edges move each edge by kd * (bounce voltage), the bounce being bmax * (number
    61	#    switching - k/2)/(k/2) (zero mean); kd in s per volt.

## Concrete defects

- **The pump does not preserve stereo frames across underruns.** In [dac.ml](/home/matthias/prog/janestreet/emulator-wt-onebit-dac/prototypes/onebit-dac/sim/dac.ml:84), bytes are fetched and sent in order, and each completed channel word commits independently at the high-byte write ([lines 188–194](/home/matthias/prog/janestreet/emulator-wt-onebit-dac/prototypes/onebit-dac/sim/dac.ml:188)). If the right-channel bytes underrun after the left word has committed, the array repeats the new left word while retaining the old right word. The policy delays/retries the missing byte; it does **not** guarantee an untorn stereo frame. E2’s repeat behavior does hold each segment’s last committed word between writes.

- **“44.1 kHz” playback is actually 44,117.6 Hz.** The 1360-clock sample interval in [dac.ml](/home/matthias/prog/janestreet/emulator-wt-onebit-dac/prototypes/onebit-dac/sim/dac.ml:19) yields 60 MHz / 1360, about 400 ppm high. The analysis also uses that rate for mode A in [dac_common.py](/home/matthias/prog/janestreet/emulator-wt-onebit-dac/prototypes/onebit-dac/analysis/dac_common.py:21). A 44.1 kHz source therefore plays slightly fast unless that rate offset is corrected.

That supports bit-exactness for those cases, not every possible input/configuration. I found no clear error in the shown rise/fall, differential, RZ, or supply-noise area equations.

- **Metrics and jitter estimate:** The THD+N and idle-noise calculations use 20 Hz–20 kHz band power; DR uses the −60 dBFS result plus 60 dB. The reported 30 ps jitter DR of about 80 dB is consistent with a first-order estimate: \(20\log_{10}(T/(\sqrt{2}\sigma_t))\) is about 81.5 dB for a 6 MHz period and 30 ps rms. Results are simulation evidence, not validation of the physical impairment assumptions.

- **Datapath and mode B:** The recorded mode A/B results support the reported performance improvement. The specific “first integrator needs 2.5–4 bits of headroom” explanation is not established by the code or results I inspected; `levels.py` sweeps input level and measures degradation, but does not calculate that headroom attribution.
